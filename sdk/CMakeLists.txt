cmake_minimum_required(VERSION 3.16)
project(d2sdk VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(D2_PACKAGE "" CACHE PATH "Fresh package produced by package.py")
if(NOT EXISTS "${D2_PACKAGE}/manifest.json")
  message(FATAL_ERROR "Run package.py first and pass -DD2_PACKAGE=/absolute/package/path")
endif()
add_library(openni SHARED IMPORTED)
set_target_properties(openni PROPERTIES IMPORTED_LOCATION "${D2_PACKAGE}/lib/libOpenNI2.so" IMPORTED_NO_SONAME TRUE)
add_library(d2sdk SHARED src/d2sdk.cpp)
target_include_directories(d2sdk PUBLIC include "${D2_PACKAGE}/include/openni")
target_compile_options(d2sdk PRIVATE -Wall -Wextra)
target_link_libraries(d2sdk PRIVATE openni)
set_target_properties(d2sdk PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${D2_PACKAGE}/lib" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
add_executable(d2_probe tools/d2_probe.cpp)
target_link_libraries(d2_probe PRIVATE d2sdk openni)
set_target_properties(d2_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${D2_PACKAGE}/bin" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
add_executable(d2_core_test tests/core_test.cpp)
target_compile_options(d2_core_test PRIVATE -UNDEBUG)
target_include_directories(d2_core_test PRIVATE include src "${D2_PACKAGE}/include/openni")
enable_testing()
add_test(NAME d2_frame_contract COMMAND d2_core_test)
